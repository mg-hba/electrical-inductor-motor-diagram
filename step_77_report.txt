  def world_to_screen(self, x, y):
          key = (round(x, 2), round(y, 2), self._version)
          c = self._cache.get(key)
          if c is not None:
              return c
          sx = (self.margin_px +
                (x - self.grid_x_min) * self.cell_size_px * self.screen_scale +
                self.offset_x)
          sy = (self.margin_px +
                (y - self.grid_y_min) * self.cell_size_px * self.screen_scale +
                self.offset_y)
          result = (sx, sy)
          self._cache[key] = result
          return result

  def screen_to_world(self, sx, sy):
          x = ((sx - self.margin_px - self.offset_x) /
               (self.cell_size_px * self.screen_scale) + self.grid_x_min)
          y = ((sy - self.margin_px - self.offset_y) /
               (self.cell_size_px * self.screen_scale) + self.grid_y_min)
          return x, y

  def handle_connection_touch_down(diagram, touch):
      if not diagram.connection.active:
          return False
      world_x, world_y = diagram.screen_to_world(*touch.pos)
      result = diagram.connection.handle_touch_down_world(
          world_x, world_y, diagram.SCREEN_scale, diagram.GRID_cell_size_px
      )
      if result:
          diagram.connection.apply_hit(result)
          diagram._rebuild_all()
          update_phase_labels(diagram)
          return True
      return False

  set_connection_mode:
  --------------------------------------------------------------------------
  def set_connection_mode(diagram, active):
      diagram.connection.set_active(active)
      if active:
          diagram.connection.load_from_json(json_manager)
      diagram._rebuild_all()
